% Shared headings and spacing. Answer/proof endings use amsthm's QED stack.
\newcommand{\lectureblockbegin}[1]{%
  \par\addvspace{4pt plus 1pt minus 1pt}%
  \normalfont\setstretch{1.75}%
  \setlength{\abovedisplayskip}{8pt plus 2pt minus 2pt}%
  \setlength{\belowdisplayskip}{8pt plus 2pt minus 2pt}%
  \noindent\hspace*{\parindent}\textbf{#1}\quad\nobreak\ignorespaces
}
\newcommand{\lectureblockend}{%
  \par\addvspace{8pt plus 2pt minus 1pt}\ignorespacesafterend
}
\NewDocumentEnvironment{analysis}{O{分析}}
  {\lectureblockbegin{#1}}
  {\lectureblockend}
\NewDocumentEnvironment{solution}{O{解答}}
  {\pushQED{\qed}\lectureblockbegin{#1}}
  {\unskip\popQED\lectureblockend}
\RenewDocumentEnvironment{proof}{O{证明}}
  {\pushQED{\qed}\lectureblockbegin{#1}}
  {\unskip\popQED\lectureblockend}
